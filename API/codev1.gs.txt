const WEBHOOK_GOOGLE_CHAT = ""

function doPost(e) {

  try {

    /*
    SEGURANÇA
    */
    if (!e || !e.postData) {

      return ContentService
        .createTextOutput("SEM POSTDATA")
    }

    /*
    JSON
    */
    const item =
      JSON.parse(e.postData.contents)

    console.log(
      JSON.stringify(item, null, 2)
    )

    /*
    SALVAR
    */
    salvarAtendimento(item)

    /*
    VALIDAR
    */
    validarCliente(item)

    return ContentService
      .createTextOutput(

        JSON.stringify({

          success: true
        })
      )
      .setMimeType(
        ContentService.MimeType.JSON
      )

  } catch (erro) {

    console.log(erro)

    return ContentService
      .createTextOutput(

        JSON.stringify({

          success: false,

          erro: erro.toString()
        })
      )
      .setMimeType(
        ContentService.MimeType.JSON
      )
  }
}


/*
==================================================
CRIAR ABA
==================================================
*/
function obterOuCriarAba(nomeAba) {

  const planilha =
    SpreadsheetApp.getActiveSpreadsheet()

  let aba =
    planilha.getSheetByName(nomeAba)

  /*
  NÃO EXISTE
  */
  if (!aba) {

    aba =
      planilha.insertSheet(nomeAba)
  }

  return aba
}


/*
==================================================
SALVAR ATENDIMENTO
==================================================
*/
function salvarAtendimento(item) {

  const aba =
    obterOuCriarAba("Dados")

  /*
  CABEÇALHO
  */
  if (aba.getLastRow() === 0) {

    aba.appendRow([

      "Data Registro",

      "Origin ID",

      "Unidade",

      "Fila",

      "Senha",

      "Cliente",

      "Gerado Em",

      "Chamado Por",

      "No Show Em",

      "Status",

      "Início Atendimento",

      "Fim Atendimento"
    ])
  }

  /*
  ORIGIN ID
  */
  const originId =
    item.origin_id || ""

  /*
  DADOS EXISTENTES
  */
  const dados =
    aba.getDataRange().getValues()

  /*
  EVITAR DUPLICIDADE
  */
  for (let i = 1; i < dados.length; i++) {

    /*
    COLUNA ORIGIN ID
    */
    if (dados[i][1] == originId) {

      console.log(
        "REGISTRO DUPLICADO"
      )

      return
    }
  }

  /*
  CHAMADO POR
  */
  const chamadoPor =

    (item.service_desk_label || "") +

    " " +

    (item.service_desk_number || "")

  /*
  SALVAR
  */
  aba.appendRow([

    new Date(),

    originId,

    item.site_label || "",

    item.queue_label || "",

    item.ticket || "",

    item.user_label || "",

    item.ticket_generated_at || "",

    chamadoPor,

    item.ticket_last_call_at || "",

    item.status || "",

    item.service_started_at || "",

    item.service_ended_at || ""
  ])
}


/*
==================================================
VALIDAR CLIENTE
==================================================
*/
function validarCliente(item) {

  /*
  SOMENTE FINALIZADOS
  */
  if (item.status != 3) {

    console.log(
      "ATENDIMENTO NÃO FINALIZADO"
    )

    return
  }

  /*
  IGNORA NO SHOW
  */
  if (item.ticket_last_call_at) {

    console.log(
      "NO SHOW IGNORADO"
    )

    return
  }

  /*
  CLIENTE
  */
  const cliente =

    String(
      item.user_label || ""
    ).trim()

  /*
  SOMENTE NÚMEROS
  */
  const regex = /^\d+$/

  let erro = ""

  /*
  CLIENTE VAZIO
  */
  if (!cliente) {

    erro = "CLIENTE VAZIO"
  }

  /*
  CLIENTE INVÁLIDO
  */
  else if (!regex.test(cliente)) {

    erro = "CLIENTE INVÁLIDO"
  }

  /*
  ALERTA
  */
  if (erro) {

    salvarAlerta(item, erro)

    enviarGoogleChat(item, erro)
  }
}


/*
==================================================
SALVAR ALERTA
==================================================
*/
function salvarAlerta(item, erro) {

  const aba =
    obterOuCriarAba("Alertas")

  /*
  CABEÇALHO
  */
  if (aba.getLastRow() === 0) {

    aba.appendRow([

      "Data",

      "Erro",

      "Origin ID",

      "Unidade",

      "Fila",

      "Senha",

      "Cliente"
    ])
  }

  /*
  SALVAR
  */
  aba.appendRow([

    new Date(),

    erro,

    item.origin_id || "",

    item.site_label || "",

    item.queue_label || "",

    item.ticket || "",

    item.user_label || ""
  ])
}


/*
==================================================
GOOGLE CHAT
==================================================
*/
function enviarGoogleChat(item, erro) {

  /*
  SEM WEBHOOK
  */
  if (

    !WEBHOOK_GOOGLE_CHAT ||

    WEBHOOK_GOOGLE_CHAT == ""

  ) {

    console.log(
      "WEBHOOK NÃO CONFIGURADO"
    )

    return
  }

  /*
  MENSAGEM
  */
  const mensagem = {

    text:

      "⚠️ ALERTA NEXTQS\n\n" +

      "Erro: " + erro + "\n\n" +

      "Unidade: " +
      (item.site_label || "") + "\n" +

      "Fila: " +
      (item.queue_label || "") + "\n" +

      "Senha: " +
      (item.ticket || "") + "\n" +

      "Cliente: " +
      (item.user_label || "")
  }

  /*
  ENVIAR
  */
  UrlFetchApp.fetch(

    WEBHOOK_GOOGLE_CHAT,

    {

      method: "post",

      contentType:
        "application/json",

      payload:
        JSON.stringify(mensagem)
    }
  )
}
