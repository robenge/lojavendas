const WEBHOOK_GOOGLE_CHAT = ""


/*
==================================================
DASHBOARD
==================================================
*/
function doGet() {

  return HtmlService
    .createTemplateFromFile("index")
    .evaluate()
    .setTitle("NEXTQS MONITOR")
    .setXFrameOptionsMode(
      HtmlService.XFrameOptionsMode.ALLOWALL
    )
}


/*
==================================================
INCLUDE HTML
==================================================
*/
function include(nomeArquivo) {

  return HtmlService
    .createHtmlOutputFromFile(nomeArquivo)
    .getContent()
}


/*
==================================================
WEBHOOK NEXTQS
==================================================
*/
function doPost(e) {

  try {

    if (!e || !e.postData) {

      return ContentService
        .createTextOutput("SEM POSTDATA")
    }

    const item =
      JSON.parse(e.postData.contents)

    console.log(
      JSON.stringify(item, null, 2)
    )

    salvarAtendimento(item)

    validarCliente(item)

    return ContentService
      .createTextOutput(

        JSON.stringify({

          success: true
        })
      )
      .setMimeType(
        ContentService.MimeType.JSON
      )

  } catch (erro) {

    console.log(erro)

    return ContentService
      .createTextOutput(

        JSON.stringify({

          success: false,

          erro: erro.toString()
        })
      )
      .setMimeType(
        ContentService.MimeType.JSON
      )
  }
}


/*
==================================================
CRIAR ABA
==================================================
*/
function obterOuCriarAba(nomeAba) {

  const planilha =
    SpreadsheetApp.getActiveSpreadsheet()

  let aba =
    planilha.getSheetByName(nomeAba)

  if (!aba) {

    aba =
      planilha.insertSheet(nomeAba)
  }

  return aba
}


/*
==================================================
SALVAR
==================================================
*/
function salvarAtendimento(item) {

  const aba =
    obterOuCriarAba("Dados")

  if (aba.getLastRow() === 0) {

    aba.appendRow([

      "Data Registro",

      "Origin ID",

      "Unidade",

      "Fila",

      "Senha",

      "Cliente",

      "Gerado Em",

      "Chamado Por",

      "No Show Em",

      "Status",

      "Início Atendimento",

      "Fim Atendimento"
    ])
  }

  const originId =
    item.origin_id || ""

  const dados =
    aba.getDataRange().getValues()

  for (let i = 1; i < dados.length; i++) {

    if (dados[i][1] == originId) {

      console.log(
        "REGISTRO DUPLICADO"
      )

      return
    }
  }

  const chamadoPor =

    (item.service_desk_label || "") +

    " " +

    (item.service_desk_number || "")

  aba.appendRow([

    new Date(),

    originId,

    item.site_label || "",

    item.queue_label || "",

    item.ticket || "",

    item.user_label || "",

    item.ticket_generated_at || "",

    chamadoPor,

    item.ticket_last_call_at || "",

    item.status || "",

    item.service_started_at || "",

    item.service_ended_at || ""
  ])
}


/*
==================================================
VALIDAR CLIENTE
==================================================
*/
function validarCliente(item) {

  if (item.status != 3) {

    return
  }

  if (item.ticket_last_call_at) {

    return
  }

  const cliente =

    String(
      item.user_label || ""
    ).trim()

  const regex = /^\d+$/

  let erro = ""

  if (!cliente) {

    erro = "CLIENTE VAZIO"
  }

  else if (!regex.test(cliente)) {

    erro = "CLIENTE INVÁLIDO"
  }

  if (erro) {

    salvarAlerta(item, erro)

    enviarGoogleChat(item, erro)
  }
}


/*
==================================================
SALVAR ALERTA
==================================================
*/
function salvarAlerta(item, erro) {

  const aba =
    obterOuCriarAba("Alertas")

  if (aba.getLastRow() === 0) {

    aba.appendRow([

      "Data",

      "Erro",

      "Origin ID",

      "Unidade",

      "Fila",

      "Senha",

      "Cliente"
    ])
  }

  aba.appendRow([

    new Date(),

    erro,

    item.origin_id || "",

    item.site_label || "",

    item.queue_label || "",

    item.ticket || "",

    item.user_label || ""
  ])
}


/*
==================================================
GOOGLE CHAT
==================================================
*/
function enviarGoogleChat(item, erro) {

  if (

    !WEBHOOK_GOOGLE_CHAT ||

    WEBHOOK_GOOGLE_CHAT == ""

  ) {

    console.log(
      "WEBHOOK NÃO CONFIGURADO"
    )

    return
  }

  const mensagem = {

    text:

      "⚠️ ALERTA NEXTQS\n\n" +

      "Erro: " + erro + "\n\n" +

      "Unidade: " +
      (item.site_label || "") + "\n" +

      "Fila: " +
      (item.queue_label || "") + "\n" +

      "Senha: " +
      (item.ticket || "") + "\n" +

      "Cliente: " +
      (item.user_label || "")
  }

  UrlFetchApp.fetch(

    WEBHOOK_GOOGLE_CHAT,

    {

      method: "post",

      contentType:
        "application/json",

      payload:
        JSON.stringify(mensagem)
    }
  )
}


/*
==================================================
BUSCAR DADOS DASHBOARD
==================================================
*/
function obterDadosDashboard() {

  const aba =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName("Dados")

  if (!aba) {

    return []
  }

  return aba
    .getDataRange()
    .getValues()
}
