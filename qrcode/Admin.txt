    let usuarioAtual = null;
    
    // ============================================

    window.onload = function() {
      let usuarioSalvo = null;
      
      // Try-catch protege caso o navegador bloqueie acesso ao localStorage
      try {
        usuarioSalvo = localStorage.getItem('usuario_admin');
      } catch (e) {
        console.warn("Acesso ao localStorage bloqueado pelo navegador.");
      }
      
      if (usuarioSalvo) {
        try {
          usuarioAtual = JSON.parse(usuarioSalvo);
          document.getElementById('userName').textContent = '👤 ' + (usuarioAtual.nome || 'Admin');
        } catch(e) {
          try { localStorage.removeItem('usuario_admin'); } catch(err){}
          redirecionarParaLogin();
          return;
        }
      } else {
        redirecionarParaLogin();
        return;
      }
      
      // Só carrega os dados se confirmou que está logado
      carregarDados();
    };

    function redirecionarParaLogin() {
      // Cria uma interface de bloqueio amigável.
      // Caso o navegador impeça o redirecionamento automático (o que causa o travamento), o botão resolve!
      document.body.innerHTML = `
        <div style="display:flex;flex-direction:column;justify-content:center;align-items:center;height:100vh;font-family:sans-serif;color:#2d3748;background:#f7fafc;text-align:center;padding:20px;">
          <div style="font-size: 50px; margin-bottom: 15px;">🔒</div>
          <h2 style="margin-bottom: 10px;">Acesso Restrito</h2>
          <p style="margin-bottom: 25px; color: #718096; max-width: 320px;">Você precisa fazer login para acessar o painel administrativo.</p>
          <a id="btnRedirecionar" href="#" target="_top" style="background: linear-gradient(135deg, #667eea 0%, #764ba2 100%); color: white; padding: 14px 28px; text-decoration: none; border-radius: 12px; font-weight: bold; box-shadow: 0 4px 15px rgba(118, 75, 162, 0.3); transition: transform 0.2s;">
            <i class="fas fa-sign-in-alt"></i> Ir para o Login
          </a>
        </div>
      `;

      if (typeof google !== 'undefined' && google.script && google.script.run) {
        google.script.run.withSuccessHandler(function(url) {
            const loginUrl = url + '?acao=login';
            const btn = document.getElementById('btnRedirecionar');
            btn.href = loginUrl;
            
            // Tenta clicar via código para redirecionar automaticamente.
            // Se o navegador bloquear o redirecionamento automático silencioso, o botão físico estará na tela.
            try { btn.click(); } catch(e) {}
        }).getUrlBase();
      } else {
        const fallbackUrl = window.location.href.split('?')[0] + '?acao=login';
        document.getElementById('btnRedirecionar').href = fallbackUrl;
      }
    }

      if (confirm('Tem certeza que deseja sair?')) {

    function mudarTab(tab) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));

      if (mapa[tab] !== undefined) botoes[mapa[tab]].classList.add('active');

    function carregarDados() {
      mostrarLoadingTabela(true);
      
      // Previne erro caso esteja rodando fora do ambiente do Google
      if (typeof google === 'undefined' || !google.script) {
        mostrarLoadingTabela(false);
        mostrarToast('Erro: Conexão com o Google Script falhou.', 'error');
        return;
      }
      
      google.script.run
        .withSuccessHandler(function(estatisticas) {
          document.getElementById('totalFrases').textContent = estatisticas.total || 0;
          document.getElementById('frasesLidas').textContent = estatisticas.lidas || 0;
          document.getElementById('surpresasRestantes').textContent = estatisticas.disponiveis || 0;
          document.getElementById('totalLeituras').textContent = estatisticas.totalLeituras || 0;
          const progresso = estatisticas.total > 0 ? Math.round((estatisticas.lidas / estatisticas.total) * 100) : 0;
          document.getElementById('progressoBarra').style.width = progresso + '%';
          document.getElementById('progressoTexto').textContent = progresso + '%';
        })
        .getEstatisticas();
      
      google.script.run
        .withSuccessHandler(function(frases) {
          mostrarLoadingTabela(false);
          renderizarTabelaFrases(frases);
        })
        .getTodasFrases();
    }
    
    function mostrarLoadingTabela(ativo) {
      document.getElementById('loadingTabela').style.display = ativo ? 'block' : 'none';
      document.getElementById('tabelaContainer').style.display = ativo ? 'none' : 'block';
    }
    
    function renderizarTabelaFrases(frases) {
      const tbody = document.getElementById('tabelaBody');
      tbody.innerHTML = '';
      if (!frases || frases.length === 0) {
        tbody.innerHTML = `<tr><td colspan="7"><div class="empty-state"><p>Nenhuma frase cadastrada.</p></div></td></tr>`;
        return;
      }
      frases.forEach(function(frase) {
        const tr = document.createElement('tr');
        let dataFormatada = '-';
        if (frase.UltimaLeitura) {
          try {
            const data = new Date(frase.UltimaLeitura);
            dataFormatada = data.toLocaleDateString('pt-BR') + ' ' + data.toLocaleTimeString('pt-BR', {hour: '2-digit', minute: '2-digit'});
          } catch(e) {}
        }
        const statusClass = frase.Status ? frase.Status.toLowerCase() : '';
        tr.innerHTML = `
          <td><strong>#${frase.ID}</strong></td>
          <td style="max-width: 200px; word-wrap: break-word;">${frase.Frase || '-'}</td>
          <td>${frase.Autor || '-'}</td>
          <td><span class="status-badge ${statusClass}">${frase.Status || 'Desconhecido'}</span></td>
          <td style="text-align: center;">${frase.Contador || 0}</td>
          <td style="font-size: 12px;">${dataFormatada}</td>
          <td>
            <div class="btn-group-actions">
              <button class="btn btn-primary btn-sm" onclick="editarFrase(${frase.ID})"><i class="fas fa-edit"></i></button>
              <button class="btn btn-success btn-sm" onclick="marcarComoLida(${frase.ID})" ${frase.Status === 'Lida' ? 'disabled' : ''}><i class="fas fa-check"></i></button>
              <button class="btn btn-danger btn-sm" onclick="excluirFrase(${frase.ID})"><i class="fas fa-trash"></i></button>
            </div>
          </td>
        `;
        tbody.appendChild(tr);
      });
    }
    
    function filtrarTabela() {
      const searchText = document.getElementById('searchInput').value.toLowerCase();
      const filterStatus = document.getElementById('filtroStatus').value;
      const rows = document.getElementById('tabelaBody').getElementsByTagName('tr');
      for (let row of rows) {
        const cells = row.getElementsByTagName('td');
        if (cells.length <= 1) continue;
        const frase = cells[1]?.textContent?.toLowerCase() || '';
        const autor = cells[2]?.textContent?.toLowerCase() || '';
        const status = cells[3]?.textContent?.trim() || '';
        const matchSearch = frase.includes(searchText) || autor.includes(searchText);
        const matchStatus = filterStatus === 'todos' || status === filterStatus;
        row.style.display = (matchSearch && matchStatus) ? '' : 'none';
      }
    }
    
    function editarFrase(id) {
      google.script.run
        .withSuccessHandler(function(frase) {
          if (frase) {
            document.getElementById('modalTitulo').textContent = '✏️ Editar Frase #' + id;
            document.getElementById('fraseId').value = id;
            document.getElementById('fraseTexto').value = frase.Frase || '';
            document.getElementById('fraseAutor').value = frase.Autor || '';
            document.getElementById('fraseStatus').value = frase.Status || 'Disponivel';
            document.getElementById('modalFrase').classList.add('active');
          }
        })
        .getFrasePorId(id);
    }
    
    function salvarFrase(event) {
      event.preventDefault();
      const dados = {
        ID: document.getElementById('fraseId').value || null,
        Frase: document.getElementById('fraseTexto').value.trim(),
        Autor: document.getElementById('fraseAutor').value.trim() || 'Desconhecido',
        Status: document.getElementById('fraseStatus').value
      };
      
      google.script.run
        .withSuccessHandler(function(resultado) {
          fecharModal('modalFrase');
          mostrarToast(resultado.mensagem, 'success');
          carregarDados();
        })
        .salvarFrase(dados);
    }
    
    function marcarComoLida(id) {
      if (!confirm('Marcar esta frase como lida?')) return;
      google.script.run.withSuccessHandler(() => { mostrarToast('✅ Marcada como lida!', 'success'); carregarDados(); }).marcarComoLida(id);
    }
    
    function excluirFrase(id) {
      if (!confirm('⚠️ Tem certeza que deseja excluir esta frase permanentemente?')) return;
      google.script.run.withSuccessHandler(() => { mostrarToast('🗑️ Frase excluída!', 'success'); carregarDados(); }).excluirFrase(id);
    }
    
    function resetarTodas() {
      if (!confirm('⚠️ Resetar TODAS as frases?')) return;
      google.script.run.withSuccessHandler(() => { mostrarToast('🔄 Resetadas!', 'success'); carregarDados(); }).resetarFrases();
    }
