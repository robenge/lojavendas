  <script>
    // ============================================
    // CORREÇÃO: Redirecionamento correto para Google Apps Script
    // ============================================
    window.onload = function() {
        var usuarioSalvo = localStorage.getItem('usuario_admin');
        if (usuarioSalvo) {
            try {
                var usuario = JSON.parse(usuarioSalvo);
                if (usuario && usuario.email) {
                    document.getElementById('btnLogin').disabled = true;
                    document.getElementById('btnLogin').innerHTML = '<span class="spinner"></span> Redirecionando...';
                    
                    google.script.run.withSuccessHandler(function(url) {
                        window.top.location.href = url + '?acao=admin';
                    }).getUrlBase();
                    return;
                }
            } catch(e) {
                localStorage.removeItem('usuario_admin');
            }
        }
    };

    function fazerLogin(event) {
        event.preventDefault();
        
        const email = document.getElementById('email').value.trim();
        const senha = document.getElementById('senha').value.trim();
        const btn = document.getElementById('btnLogin');
        const erro = document.getElementById('erroLogin');
        
        if (!email || !senha) {
            mostrarToast('Preencha todos os campos', 'error');
            return;
        }
        
        btn.disabled = true;
        btn.innerHTML = '<span class="spinner"></span> Autenticando...';
        erro.classList.remove('show');
        
        google.script.run
            .withSuccessHandler(function(resultado) {
                if (resultado.sucesso) {
                    mostrarToast('✅ Login realizado com sucesso!', 'success');
                    localStorage.setItem('usuario_admin', JSON.stringify(resultado.usuario));
                    
                    // CORREÇÃO: Usa a URL base do Google Scripts e redireciona a janela inteira
                    google.script.run.withSuccessHandler(function(url) {
                        window.top.location.href = url + '?acao=admin';
                    }).getUrlBase();
                } else {
                    erro.textContent = '❌ ' + (resultado.mensagem || 'Erro desconhecido');
                    erro.classList.add('show');
                    btn.disabled = false;
                    btn.innerHTML = '<i class="fas fa-sign-in-alt"></i> Entrar';
                }
            })
            .withFailureHandler(function(erroMsg) {
                mostrarToast('Erro ao fazer login. Servidor indisponível.', 'error');
                btn.disabled = false;
                btn.innerHTML = '<i class="fas fa-sign-in-alt"></i> Entrar';
            })
            .autenticarUsuario(email, senha);
    }
    
    function mostrarToast(mensagem, tipo = 'info') {
        const toast = document.getElementById('toast');
        toast.textContent = mensagem;
        toast.className = 'toast show ' + (tipo === 'error' ? '' : 'success');
        if (tipo === 'error') toast.style.background = '#fc8181';
        else toast.style.background = '#48bb78';
        setTimeout(() => toast.className = 'toast', 3000);
    }
