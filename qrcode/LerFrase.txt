  <script>
    // ============================================
    // VARIÁVEIS
    // ============================================
    let fraseAtual = null;
    let jaLeu = false;
    let totalFrases = 0;
    let lidas = 0;
    
    // ============================================
    // CARREGAR FRASE
    // ============================================
    function carregarFrase() {
      google.script.run
        .withSuccessHandler(function(frase) {
          if (frase && !frase.erro) {
            fraseAtual = frase;
            atualizarUI(frase);
            
            // Revelar com animação
            setTimeout(() => {
              document.getElementById('cardMensagem').classList.add('revelada');
              criarConfetes();
            }, 300);
          } else {
            document.getElementById('frase').textContent = '🎉 Todas as mensagens foram descobertas!';
            document.getElementById('autor').textContent = 'Volte em breve para mais surpresas';
            document.getElementById('btnLer').disabled = true;
          }
        })
        .withFailureHandler(function(erro) {
          mostrarToast('Erro ao carregar mensagem', 'error');
          console.error(erro);
        })
        .pegarProximaFrase();
    }
    
    // ============================================
    // ATUALIZAR UI
    // ============================================
    function atualizarUI(frase) {
      document.getElementById('frase').textContent = frase.frase;
      document.getElementById('autor').textContent = frase.autor;
      document.getElementById('contador').textContent = frase.contador || 0;
      document.getElementById('numeroFrase').textContent = `#${frase.id || 1}`;
      
      // Habilitar botão
      document.getElementById('btnLer').disabled = false;
      document.getElementById('btnLer').style.display = 'flex';
      document.getElementById('btnProxima').style.display = 'none';
      
      // Icone animado
      document.getElementById('iconAnimado').textContent = '🎁';
    }
    
    // ============================================
    // LER FRASE (MARCAR COMO LIDA)
    // ============================================
    function lerFrase() {
      if (!fraseAtual || jaLeu) return;
      
      const btn = document.getElementById('btnLer');
      btn.disabled = true;
      btn.innerHTML = '<span class="spinner"></span> Processando...';
      
      google.script.run
        .withSuccessHandler(function(resultado) {
          jaLeu = true;
          
          // Animação de sucesso
          document.getElementById('iconAnimado').textContent = '🎉';
          document.getElementById('cardMensagem').style.borderColor = '#48bb78';
          
          // Confetes
          criarConfetes();
          
          mostrarToast('🎉 Mensagem descoberta com sucesso!', 'success');
          
          // Mudar botões
          btn.style.display = 'none';
          document.getElementById('btnProxima').style.display = 'flex';
          document.getElementById('btnProxima').innerHTML = '<i class="fas fa-gift"></i> Próxima Surpresa 🎁';
          
          // Atualizar contador local
          document.getElementById('contador').textContent = (fraseAtual.contador || 0) + 1;
          
        })
        .withFailureHandler(function(erro) {
          mostrarToast('Erro ao processar leitura', 'error');
          console.error(erro);
          btn.disabled = false;
          btn.innerHTML = '<i class="fas fa-check-circle"></i> Li esta mensagem! ✅';
        })
        .lerFrase();
    }
    
    // ============================================
    // PRÓXIMA SURPRESA
    // ============================================
    function proximaFrase() {
      // Resetar UI
      jaLeu = false;
      document.getElementById('btnLer').style.display = 'flex';
      document.getElementById('btnProxima').style.display = 'none';
      document.getElementById('btnLer').disabled = true;
      document.getElementById('btnLer').innerHTML = '<span class="spinner"></span> Carregando...';
      document.getElementById('iconAnimado').textContent = '🔄';
      document.getElementById('cardMensagem').classList.remove('revelada');
      document.getElementById('cardMensagem').style.borderColor = 'transparent';
      
      // Carregar nova frase
      setTimeout(carregarFrase, 500);
    }
    
    // ============================================
    // CRIAR CONFETES
    // ============================================
    function criarConfetes() {
      const container = document.createElement('div');
      container.className = 'confetti-container';
      document.body.appendChild(container);
      
      const cores = ['#667eea', '#764ba2', '#f093fb', '#f5576c', '#4facfe', '#43e97b', '#fa709a', '#f9ca24'];
      const emojis = ['🎉', '✨', '🎊', '⭐', '💫', '🌟', '🎈', '🎁', '💝'];
      
      for (let i = 0; i < 40; i++) {
        const confetti = document.createElement('div');
        confetti.className = 'confetti';
        confetti.textContent = emojis[Math.floor(Math.random() * emojis.length)];
        confetti.style.left = Math.random() * 100 + '%';
        confetti.style.fontSize = (Math.random() * 20 + 10) + 'px';
        confetti.style.animationDuration = (Math.random() * 2 + 2) + 's';
        confetti.style.animationDelay = (Math.random() * 2) + 's';
        container.appendChild(confetti);
      }
      
      setTimeout(() => {
        container.remove();
      }, 5000);
    }
    
    // ============================================
    // ATUALIZAR ESTATÍSTICAS
    // ============================================
    function atualizarEstatisticas() {
      google.script.run
        .withSuccessHandler(function(estatisticas) {
          if (estatisticas) {
            totalFrases = estatisticas.total || 0;
            lidas = estatisticas.lidas || 0;
            document.getElementById('totalLidas').textContent = lidas;
          }
        })
        .withFailureHandler(function(erro) {
          console.error('Erro ao carregar estatísticas:', erro);
        })
        .getEstatisticas();
    }
    
    // ============================================
    // TOAST
    // ============================================
    function mostrarToast(mensagem, tipo = 'info') {
      const toast = document.getElementById('toast');
      toast.textContent = mensagem;
      toast.className = 'toast show';
      
      if (tipo === 'error') {
        toast.style.background = '#fc8181';
      } else if (tipo === 'success') {
        toast.style.background = '#48bb78';
      } else {
        toast.style.background = '#4299e1';
      }
      
      setTimeout(() => {
        toast.className = 'toast';
      }, 3000);
    }
    
    // ============================================
    // ATUALIZAR DATA
    // ============================================
    function atualizarData() {
      const agora = new Date();
      const opcoes = { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' };
      document.getElementById('dataAtual').textContent = '📅 ' + agora.toLocaleDateString('pt-BR', opcoes);
    }
    
    // ============================================
    // INICIALIZAÇÃO
    // ============================================
    window.onload = function() {
      carregarFrase();
      atualizarData();
      setTimeout(atualizarEstatisticas, 1000);
      
      document.getElementById('btnLer').addEventListener('click', lerFrase);
      document.getElementById('btnProxima').addEventListener('click', proximaFrase);
      
      // Atualizar a cada 30 segundos
      setInterval(atualizarData, 30000);
    };
  </script>
