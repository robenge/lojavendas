const PLANILHA = SpreadsheetApp.getActiveSpreadsheet();
const ABA_FRASES = PLANILHA.getSheetByName('Frases');

// ============================================
// FUNÇÃO: Buscar Frase Atual
// ============================================
function buscarFraseAtual() {
  const dados = ABA_FRASES.getDataRange().getValues();
  const cabecalho = dados[0];
  
  let idxID = cabecalho.indexOf('ID');
  let idxFrase = cabecalho.indexOf('Frase');
  let idxAutor = cabecalho.indexOf('Autor');
  let idxStatus = cabecalho.indexOf('Status');
  let idxContador = cabecalho.indexOf('Contador');
  
  // Buscar primeira frase disponível
  for (let i = 1; i < dados.length; i++) {
    if (dados[i][idxStatus] === 'Disponivel') {
      return {
        id: dados[i][idxID],
        frase: dados[i][idxFrase],
        autor: dados[i][idxAutor],
        status: dados[i][idxStatus],
        contador: dados[i][idxContador],
        linha: i + 1
      };
    }
  }
  
  // Se todas lidas, resetar
  resetarFrases();
  
  // Buscar novamente após reset
  for (let i = 1; i < dados.length; i++) {
    if (dados[i][idxStatus] === 'Disponivel') {
      return {
        id: dados[i][idxID],
        frase: dados[i][idxFrase],
        autor: dados[i][idxAutor],
        status: dados[i][idxStatus],
        contador: dados[i][idxContador],
        linha: i + 1
      };
    }
  }
  
  return null;
}

// ============================================
// FUNÇÃO: Marcar como Lida
// ============================================
function marcarComoLida(idFrase) {
  const dados = ABA_FRASES.getDataRange().getValues();
  const cabecalho = dados[0];
  
  let idxID = cabecalho.indexOf('ID');
  let idxStatus = cabecalho.indexOf('Status');
  let idxUltimaLeitura = cabecalho.indexOf('UltimaLeitura');
  let idxContador = cabecalho.indexOf('Contador');
  
  for (let i = 1; i < dados.length; i++) {
    if (dados[i][idxID] == idFrase) {
      ABA_FRASES.getRange(i + 1, idxStatus + 1).setValue('Lida');
      ABA_FRASES.getRange(i + 1, idxUltimaLeitura + 1).setValue(new Date());
      ABA_FRASES.getRange(i + 1, idxContador + 1).setValue(dados[i][idxContador] + 1);
      return true;
    }
  }
  return false;
}

// ============================================
// FUNÇÃO: Resetar Frases
// ============================================
function resetarFrases() {
  const dados = ABA_FRASES.getDataRange().getValues();
  const cabecalho = dados[0];
  let idxStatus = cabecalho.indexOf('Status');
  
  for (let i = 1; i < dados.length; i++) {
    ABA_FRASES.getRange(i + 1, idxStatus + 1).setValue('Disponivel');
  }
}

// ============================================
// FUNÇÃO: Pegar Próxima Frase (para API)
// ============================================
function pegarProximaFrase() {
  let frase = buscarFraseAtual();
  if (!frase) {
    resetarFrases();
    frase = buscarFraseAtual();
  }
  return frase;
}

// ============================================
// FUNÇÃO: Ler Frase (Ação Principal)
// ============================================
function lerFrase() {
  // Buscar frase atual
  let fraseAtual = buscarFraseAtual();
  
  if (!fraseAtual) {
    return { erro: 'Nenhuma frase disponível' };
  }
  
  // Marcar como lida
  marcarComoLida(fraseAtual.id);
  
  // Buscar próxima frase para exibir
  let proximaFrase = buscarFraseAtual();
  
  // Retornar a próxima frase (já marcada como lida)
  return {
    mensagem: '✅ Frase lida com sucesso!',
    proximaFrase: proximaFrase,
    fraseLida: fraseAtual
  };
}

// ============================================
// FUNÇÃO: Estatísticas
// ============================================
function getEstatisticas() {
  const dados = ABA_FRASES.getDataRange().getValues();
  const cabecalho = dados[0];
  let idxStatus = cabecalho.indexOf('Status');
  let idxContador = cabecalho.indexOf('Contador');
  
  let total = dados.length - 1;
  let disponiveis = 0;
  let lidas = 0;
  let totalLeituras = 0;
  
  for (let i = 1; i < dados.length; i++) {
    if (dados[i][idxStatus] === 'Disponivel') {
      disponiveis++;
    } else if (dados[i][idxStatus] === 'Lida') {
      lidas++;
    }
    totalLeituras += dados[i][idxContador] || 0;
  }
  
  return { 
    total, 
    disponiveis, 
    lidas, 
    totalLeituras 
  };
}

// Buscar todas as frases
function getTodasFrases() {

  return frases;
}

// Salvar frase (nova ou edição)
function salvarFrase(dados) {
  const id = dados.ID;
  const frase = dados.Frase;
  const autor = dados.Autor || 'Desconhecido';
  const status = dados.Status || 'Disponivel';
  
  const sheet = ABA_FRASES;
  const ultimaLinha = sheet.getLastRow();
  const cabecalho = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  
  let idxID = cabecalho.indexOf('ID');
  let idxFrase = cabecalho.indexOf('Frase');
  let idxAutor = cabecalho.indexOf('Autor');
  let idxStatus = cabecalho.indexOf('Status');
  
  if (id) {
    // Editar frase existente
    for (let i = 2; i <= ultimaLinha; i++) {
      const idAtual = sheet.getRange(i, idxID + 1).getValue();
      if (idAtual == id) {
        sheet.getRange(i, idxFrase + 1).setValue(frase);
        sheet.getRange(i, idxAutor + 1).setValue(autor);
        sheet.getRange(i, idxStatus + 1).setValue(status);
        return { mensagem: '✅ Frase atualizada com sucesso!', id: id };
      }
    }
    return { mensagem: '❌ Frase não encontrada', erro: true };
  } else {
    // Nova frase - gerar novo ID
    let novoId = 1;
    for (let i = 2; i <= ultimaLinha; i++) {
      const idAtual = sheet.getRange(i, idxID + 1).getValue();
      if (idAtual >= novoId) novoId = idAtual + 1;
    }
    
    sheet.appendRow([novoId, frase, autor, status, '', 0]);
    return { mensagem: '✅ Nova frase adicionada com sucesso!', id: novoId };
  }
}

// Excluir frase
function excluirFrase(id) {
  const dados = ABA_FRASES.getDataRange().getValues();
  const cabecalho = dados[0];
  let idxID = cabecalho.indexOf('ID');
  
  for (let i = dados.length - 1; i >= 1; i--) {
    if (dados[i][idxID] == id) {
      ABA_FRASES.deleteRow(i + 1);
      return { mensagem: '🗑️ Frase excluída com sucesso!' };
    }
  }
  
  return { mensagem: '❌ Frase não encontrada', erro: true };
}

// FUNÇÃO: Autenticar Usuário (Login)
// ============================================
function autenticarUsuario(email, senha) {
  try {
    const abaUsuarios = PLANILHA.getSheetByName('Usuarios');
    
    // Fallback: se não tiver aba Usuarios, criar login padrão para você não ficar trancado de fora
    if (!abaUsuarios) {
      if (email === 'admin@admin.com' && senha === 'admin') {
        return { sucesso: true, usuario: { id: 1, nome: 'Admin', email: email, nivel: 'admin' } };
      }
      return { sucesso: false, mensagem: 'A aba "Usuarios" não existe na planilha.' };
    }
    
    const dados = abaUsuarios.getDataRange().getValues();
    const cabecalho = dados[0];
    
    let idxID = cabecalho.indexOf('ID');
    let idxNome = cabecalho.indexOf('Nome');
    let idxEmail = cabecalho.indexOf('Email');
    let idxSenha = cabecalho.indexOf('Senha');
    let idxNivel = cabecalho.indexOf('Nivel');
    let idxStatus = cabecalho.indexOf('Status');
    
    for (let i = 1; i < dados.length; i++) {
      if (String(dados[i][idxEmail]).trim() === email.trim() && String(dados[i][idxSenha]).trim() === senha.trim()) {
        if (dados[i][idxStatus] !== 'Ativo') {
          return { sucesso: false, mensagem: 'Usuário inativo.' };
        }
        return { 
          sucesso: true, 
          usuario: { 
            id: dados[i][idxID], 
            nome: dados[i][idxNome], 
            email: dados[i][idxEmail], 
            nivel: dados[i][idxNivel] 
          } 
        };
      }
    }
    return { sucesso: false, mensagem: 'Email ou senha incorretos.' };
  } catch(e) {
    return { sucesso: false, mensagem: 'Erro interno: ' + e.message };
  }
}

// FUNÇÃO: doGet (Roteamento de Páginas)
// ============================================
function doGet(e) {
  const params = e && e.parameter;
  
  // Rota para Admin
  if (params && params.acao === 'admin') {
    return HtmlService.createHtmlOutputFromFile('Admin')
      .setTitle('⚙️ Painel Admin')
      .setFaviconUrl('https://cdn.jsdelivr.net/npm/emoji-datasource-apple@14.0.0/img/apple/64/2699.png')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  
  // Nova Rota para Login
  if (params && params.acao === 'login') {
    return HtmlService.createHtmlOutputFromFile('Login')
      .setTitle('🔐 Login Admin')
      .setFaviconUrl('https://cdn.jsdelivr.net/npm/emoji-datasource-apple@14.0.0/img/apple/64/1f512.png')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  
  // Rota para Ler Frase (celular)
  if (params && params.acao === 'ler') {
    return HtmlService.createHtmlOutputFromFile('LerFrase')
      .setTitle('🎁 Mensagem Surpresa')
      .setFaviconUrl('https://cdn.jsdelivr.net/npm/emoji-datasource-apple@14.0.0/img/apple/64/1f4aa.png')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  
  // Rota Principal (QR Code)
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('📱 QR Code Motivacional')
    .setFaviconUrl('https://cdn.jsdelivr.net/npm/emoji-datasource-apple@14.0.0/img/apple/64/1f4aa.png')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// FUNÇÃO: Menu Personalizado no Google Sheets
// ============================================
function onOpen() {
  const ui = SpreadsheetApp.getUi();
  ui.createMenu('📱 Mensagens Motivacionais')
    .addItem('📊 Ver Estatísticas', 'mostrarEstatisticas')
    .addItem('🔄 Resetar Frases', 'resetarFrases')
    .addItem('➕ Adicionar Frase Padrão', 'adicionarFrasePadrao')
    .addSeparator()
    .addItem('📋 Ir para Admin', 'abrirAdmin')
    .addToUi();
}

function mostrarEstatisticas() {
  const stats = getEstatisticas();
  const ui = SpreadsheetApp.getUi();
  ui.alert('📊 Estatísticas', 
    `📚 Total de frases: ${stats.total}\n` +
    `🎁 Disponíveis: ${stats.disponiveis}\n` +
    `✅ Já lidas: ${stats.lidas}\n` +
    `👀 Total de leituras: ${stats.totalLeituras}`, 
    ui.ButtonSet.OK
  );
}

function adicionarFrasePadrao() {
  const ui = SpreadsheetApp.getUi();
  const response = ui.prompt('➕ Nova Frase', 'Digite a frase e o autor (separados por | ):', ui.ButtonSet.OK_CANCEL);
  
  if (response.getSelectedButton() === ui.Button.OK) {
    const texto = response.getResponseText();
    const partes = texto.split('|');
    const frase = partes[0]?.trim() || 'Nova frase motivacional';
    const autor = partes[1]?.trim() || 'Desconhecido';
    
    const dados = {
      ID: null,
      Frase: frase,
      Autor: autor,
      Status: 'Disponivel'
    };
    
    const resultado = salvarFrase(dados);
    ui.alert(resultado.mensagem);
  }
}

function abrirAdmin() {
  const url = ScriptApp.getService().getUrl() + '?acao=admin';
  const html = `<html><body><script>window.open('${url}', '_blank');google.script.host.close();</script></body></html>`;
  SpreadsheetApp.getUi().showModalDialog(HtmlService.createHtmlOutput(html), 'Abrindo Admin...');
}
