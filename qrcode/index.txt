  <script>
    // ============================================
    // CONFIGURAÇÃO
    // ============================================
    let urlBase = '';
    let qrCodeAtual = null;
    
    // ============================================
    // GERAR QR CODE
    // ============================================
    function gerarQRCode() {
      google.script.run
        .withSuccessHandler(function(url) {
          urlBase = url;
          const urlComParametro = url + '?acao=ler';
          
          document.getElementById('qrcode').innerHTML = '';
          
          qrCodeAtual = new QRCode(document.getElementById('qrcode'), {
            text: urlComParametro,
            width: 200,
            height: 200,
            colorDark: '#2d3748',
            colorLight: '#ffffff',
            correctLevel: QRCode.CorrectLevel.H
          });
          
          mostrarToast('✅ QR Code atualizado!', 'info');
        })
        .withFailureHandler(function(erro) {
          mostrarToast('Erro ao gerar QR Code', 'error');
          console.error(erro);
        })
        .getUrlBase();
    }
    
    // ============================================
    // ABRIR ADMIN
    // ============================================
    function abrirAdmin() {
        // Use a função do Google Apps Script para obter a URL base correta
        google.script.run
            .withSuccessHandler(function(urlBase) {
                // Constrói a URL completa com o parâmetro para o admin
                const urlAdmin = urlBase + '?acao=admin';
                // Abre em uma nova aba/janela
                window.open(urlAdmin, '_blank');
            })
            .withFailureHandler(function(erro) {
                console.error('Erro ao obter URL base:', erro);
                alert('Erro ao abrir o painel admin. Verifique o console para mais detalhes.');
            })
            .getUrlBase(); // Chama a função do backend
    }
    
    // ============================================
    // ATUALIZAR DATA
    // ============================================
    function atualizarData() {
      const agora = new Date();
      const opcoes = { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' };
      document.getElementById('dataAtual').textContent = '📅 ' + agora.toLocaleDateString('pt-BR', opcoes);
    }
    
    // ============================================
    // TOAST
    // ============================================
    function mostrarToast(mensagem, tipo = 'info') {
      const toast = document.getElementById('toast');
      toast.textContent = mensagem;
      toast.className = 'toast show';
      
      if (tipo === 'error') {
        toast.className = 'toast show error';
      } else if (tipo === 'info') {
        toast.className = 'toast show info';
      }
      
      setTimeout(function() {
        toast.className = 'toast';
      }, 2500);
    }
    
    // ============================================
    // ATUALIZAR QR CODE (com animação)
    // ============================================
    function atualizarQRCode() {
      const container = document.getElementById('qrcode');
      container.innerHTML = '';
      container.textContent = '⏳';
      container.style.opacity = '0.5';
      
      setTimeout(function() {
        gerarQRCode();
        container.style.opacity = '1';
        mostrarToast('🔄 QR Code atualizado!', 'info');
      }, 500);
    }
    
    // ============================================
    // INICIALIZAÇÃO
    // ============================================
    window.onload = function() {
      // Gerar QR Code
      setTimeout(gerarQRCode, 300);
      
      // Atualizar data
      atualizarData();
      
      // Atualizar automaticamente a cada 5 minutos (caso a URL mude)
      setInterval(function() {
        gerarQRCode();
      }, 300000);
      
      // Atualizar data a cada minuto
      setInterval(atualizarData, 60000);
      
      // Keyboard shortcut: R para atualizar
      document.addEventListener('keydown', function(e) {
        if (e.key === 'r' || e.key === 'R') {
          if (!e.ctrlKey && !e.metaKey) {
            e.preventDefault();
            atualizarQRCode();
          }
        }
      });
    };
  </script>
